    let audioCtx = null;
    function playTone(freq, type = 'sine', duration = 0.12, gainVal = 0.14, slideFreq = null) {
      if (!state.soundOn) return;
      try {
        if (!audioCtx) {
          const Ctx = window.AudioContext || window.webkitAudioContext;
          if (Ctx) audioCtx = new Ctx();
        }
        if (!audioCtx) return;
        if (audioCtx.state === 'suspended') audioCtx.resume();

        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = type;
        osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
        if (slideFreq) {
          osc.frequency.exponentialRampToValueAtTime(slideFreq, audioCtx.currentTime + duration);
        }
        gain.gain.setValueAtTime(gainVal, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + duration);
      } catch (err) {
        // Ignore audio errors on restricted frames
      }
    }

    function spawnParticles(x, y, count = 14, color = '#fb923c', speed = 16, size = 4) {
      for (let i = 0; i < count; i++) {
        const ang = Math.random() * Math.PI * 2;
        const spd = (0.35 + Math.random() * 0.85) * speed * 14;
        state.particles.push({
          x,
          y,
          vx: Math.cos(ang) * spd,
          vy: Math.sin(ang) * spd - 60,
          color,
          size: size * (0.6 + Math.random() * 0.7),
          life: 0.35 + Math.random() * 0.3,
          maxLife: 0.65
        });
      }
    }

    }

    function savePreferences() {
      try {
        localStorage.setItem('nba_jam_2026_prefs', JSON.stringify({
          playerTeamId: state.playerTeamId,
          cpuTeamId: state.cpuTeamId,
          cpuDiff: state.cpuDiff,
          mathDiff: state.mathDiff,
          soundOn: state.soundOn
        }));

    }

    function loadPreferences() {
      try {
        const raw = localStorage.getItem('nba_jam_2026_prefs');
        if (!raw) return;
        const p = JSON.parse(raw);
        if (p.playerTeamId && NBA_TEAMS.some(t => t.id === p.playerTeamId)) state.playerTeamId = p.playerTeamId;
        if (p.cpuTeamId && NBA_TEAMS.some(t => t.id === p.cpuTeamId)) state.cpuTeamId = p.cpuTeamId;
        if (['rookie', 'allstar', 'legend'].includes(p.cpuDiff)) state.cpuDiff = p.cpuDiff;
        if (['easy', 'medium', 'hard'].includes(p.mathDiff)) state.mathDiff = p.mathDiff;
        if (typeof p.soundOn === 'boolean') state.soundOn = p.soundOn;
      } catch (e) {}
    }

    // 5. QUADRATIC FACTORING ENGINE (EASY / MEDIUM / HARD)
    // =========================================================================
    function formatFactor(a, b) {
      const xPart = a === 1 ? 'x' : `${a}x`;
      if (b === 0) return `(${xPart})`;
      return b > 0 ? `(${xPart} + ${b})` : `(${xPart} - ${Math.abs(b)})`;
    }

    function formatQuadraticExpr(A, B, C) {
      let expr = A === 1 ? 'x²' : `${A}x²`;
      if (B !== 0) {
        if (B === 1) expr += ' + x';
        else if (B === -1) expr += ' - x';
        else if (B > 0) expr += ` + ${B}x`;
        else expr += ` - ${Math.abs(B)}x`;
      }
      if (C !== 0) {
        expr += C > 0 ? ` + ${C}` : ` - ${Math.abs(C)}`;
      }
      return expr;
    }

    function gcd(a, b) {
      a = Math.abs(a);
      b = Math.abs(b);
      while (b) {
        const t = b;
        b = a % b;
        a = t;
      }
      return a || 1;
    }

    function generateQuadraticProblem(tier) {
      let p = 1, q = 0, r = 1, s = 0;
      let A = 1, B = 0, C = 0;

      for (let attempts = 0; attempts < 80; attempts++) {
        if (tier === 'easy') {
          p = 1;
          r = 1;
          q = (Math.floor(Math.random() * 8) + 1) * (Math.random() < 0.5 ? 1 : -1);
          s = (Math.floor(Math.random() * 8) + 1) * (Math.random() < 0.5 ? 1 : -1);
          A = 1;
          B = q + s;
          C = q * s;
          if (Math.abs(A * C) < 70 && C !== 0) break;
        } else if (tier === 'medium') {
          p = 1;
          r = 1;
          q = (Math.floor(Math.random() * 11) + 2) * (Math.random() < 0.5 ? 1 : -1);
          s = (Math.floor(Math.random() * 11) + 2) * (Math.random() < 0.5 ? 1 : -1);
          A = 1;
          B = q + s;
          C = q * s;
          if (Math.abs(A * C) >= 24 && Math.abs(A * C) < 100 && C !== 0) break;
        } else {
          const leadChoices = [1, 2, 2, 3, 3, 4];
          p = leadChoices[Math.floor(Math.random() * leadChoices.length)];
          r = p === 1 ? [2, 3][Math.floor(Math.random() * 2)] : 1;
          q = (Math.floor(Math.random() * 7) + 1) * (Math.random() < 0.5 ? 1 : -1);
          s = (Math.floor(Math.random() * 7) + 1) * (Math.random() < 0.5 ? 1 : -1);
          if (gcd(p, q) !== 1 || gcd(r, s) !== 1) continue;
          A = p * r;
          B = p * s + q * r;
          C = q * s;
          if (A >= 1 && Math.abs(A * C) <= 120 && C !== 0) break;
        }
      }

      const f1 = formatFactor(p, q);
      const f2 = formatFactor(r, s);
      const correctStr = f1 === f2 ? `${f1}²` : `${f1}${f2}`;

      const distractors = new Set();
      const cand1 = `${formatFactor(p, -q)}${formatFactor(r, -s)}`;
      const cand2 = `${formatFactor(p, -q)}${formatFactor(r, s)}`;
      const cand3 = `${formatFactor(p, q)}${formatFactor(r, -s)}`;
      const cand4 = `${formatFactor(r, q)}${formatFactor(p, s)}`;
      const cand5 = `${formatFactor(p, q + 1)}${formatFactor(r, s - 1)}`;
      const cand6 = `${formatFactor(p, q - 1)}${formatFactor(r, s + 1)}`;

      [cand1, cand2, cand3, cand4, cand5, cand6].forEach(c => {
        if (c !== correctStr) distractors.add(c);
      });

      while (distractors.size < 3) {
        const dq = q + (Math.random() < 0.5 ? 2 : -2) || 3;
        const cand = `${formatFactor(p, dq)}${formatFactor(r, s)}`;
        if (cand !== correctStr) distractors.add(cand);
      }

      const choices = [correctStr, ...Array.from(distractors).slice(0, 3)];
      for (let i = choices.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [choices[i], choices[j]] = [choices[j], choices[i]];
      }

      return {
        expr: formatQuadraticExpr(A, B, C),
        correct: correctStr,
        choices
      };
    }

    function selectMultipleChoice(selectedChoice, clickedBtn) {
      if (state.mathAnswered) return;
      state.mathAnswered = true;
      state.mathTotal++;

      const prob = state.currentProblem;
      const isCorrect = selectedChoice === prob.correct;

      // Single-attempt lockout: disable all 4 buttons and reveal the right/wrong choices
      document.querySelectorAll('.math-choice-btn').forEach(btn => {
        btn.disabled = true;
        btn.classList.remove('hover:bg-slate-800', 'cursor-pointer');
        btn.classList.add('cursor-not-allowed');
        const text = prob.choices[Number(btn.dataset.choiceIdx)];
        if (text === prob.correct) {
          btn.className = 'math-choice-btn py-3.5 px-4 rounded-xl bg-emerald-950/90 border-2 border-emerald-400 font-arcade text-lg sm:text-xl text-emerald-300 tracking-wide';
        } else if (btn === clickedBtn && !isCorrect) {
          btn.className = 'math-choice-btn py-3.5 px-4 rounded-xl bg-rose-950/90 border-2 border-rose-500 font-arcade text-lg sm:text-xl text-rose-300 tracking-wide';
        } else {
          btn.className = 'math-choice-btn py-3.5 px-4 rounded-xl bg-[#090d18] border-2 border-slate-800 font-arcade text-lg sm:text-xl text-slate-500 tracking-wide opacity-60';
        }
      });

      const feedbackBox = document.getElementById('mathFeedbackBox');
      feedbackBox.classList.remove('hidden');

      if (isCorrect) {
        state.mathCorrect++;
        state.playerTurbo = 100;
        state.playerStreak = Math.min(CONFIG.ON_FIRE_REQ, state.playerStreak + 1);
        feedbackBox.className = 'rounded-xl p-3 border border-emerald-500/60 bg-emerald-950/70 text-emerald-200 text-sm font-bold mb-3';
        feedbackBox.textContent = `SWISH! ${prob.expr} = ${prob.correct} — Full Turbo & Streak Boost earned!`;
        playTone(523.25, 'sine', 0.14, 0.16, 783.99);
      } else {
        state.playerTurbo = Math.max(0, state.playerTurbo * 0.5);
        const activeUserPlayers = [state.players[0], state.players[1]].filter(p => p && !p.fouledOut);
        const targetPlayer = (state.players[state.controlledIdx] && !state.players[state.controlledIdx].fouledOut)
          ? state.players[state.controlledIdx]
          : activeUserPlayers[0];

        let foulMsg = 'Squad stamina reduced by 50%.';
        if (targetPlayer) {
          targetPlayer.fouls = (targetPlayer.fouls || 0) + 1;
          if (targetPlayer.fouls >= 4) {
            targetPlayer.fouledOut = true;
            foulMsg = `${targetPlayer.name} picked up Foul #4 and FOULED OUT!`;
            const remainingIdx = [0, 1].find(i => state.players[i] && !state.players[i].fouledOut);
            if (remainingIdx !== undefined) {
              state.controlledIdx = remainingIdx;
              if (state.ball.owner === targetPlayer.idx) {
                state.ball.owner = remainingIdx;
              }
              foulMsg += ` Playing 1-on-2 with ${state.players[remainingIdx].name}!`;
            } else {
              state.forfeitLoss = true;
              foulMsg += ' Both players fouled out — FORFEIT LOSS!';
            }
          } else {
            foulMsg = `FOUL on ${targetPlayer.name} (${targetPlayer.fouls}/4 fouls)!`;
          }
        }

        feedbackBox.className = 'rounded-xl p-3 border border-rose-500/60 bg-rose-950/70 text-rose-200 text-sm font-bold mb-3';
        feedbackBox.textContent = `BRICK! Correct: ${prob.correct} — ${foulMsg}`;
        playTone(170, 'sawtooth', 0.25, 0.15, 110);
      }

      updateHud();
      document.getElementById('mathNextBtn').classList.remove('hidden');
    }

      document.getElementById('hudQuarter').textContent = state.quarter <= 4 ? `Q${state.quarter}` : 'OT';
      document.getElementById('hudMathTimer').textContent = `TIMEOUT IN ${Math.max(0, Math.ceil(state.mathTimer))}s`;
    }

    // =========================================================================
    // 7. PLAYER ACTIONS (SHOOT / DUNK / BLOCK / PASS / ALLEY-OOP / STEAL / SWITCH)
    // =========================================================================
    function handlePlayerAction(action) {
      if (state.mode !== 'PLAYING') return;
      const me = state.players[state.controlledIdx];
      if (!me || me.knockdownTimer > 0) return;

      const hasBall = !state.ball.inFlight && state.ball.owner === state.controlledIdx;
      const teamHasBall = !state.ball.inFlight && (state.ball.owner === 0 || state.ball.owner === 1);

      if (action === 'shoot') {
        if (hasBall && me.z === 0) {
          launchShot(state.controlledIdx);
        } else if (!teamHasBall && me.z === 0) {
          triggerBlockJump(state.controlledIdx);
        }
      } else if (action === 'pass') {
        const isTurboHeld = Boolean(keys['ShiftLeft'] || keys['ShiftRight']);
        const mateIdx = state.controlledIdx === 0 ? 1 : 0;
        const mateAvailable = state.players[mateIdx] && !state.players[mateIdx].fouledOut;

        if (hasBall && me.z === 0) {
          if (!mateAvailable) {
            triggerAnnouncer('NO TEAMMATE AVAILABLE!', 0.9);
            return;
          }
          if (isTurboHeld) {
            launchAlleyOop(state.controlledIdx, mateIdx);
          } else {
            launchPass(state.controlledIdx, mateIdx);
          }
        } else if (teamHasBall && !hasBall) {
          if (mateAvailable && state.players[mateIdx].z === 0 && me.z === 0) {
            if (isTurboHeld) {
              launchAlleyOop(mateIdx, state.controlledIdx);
            } else {
              launchPass(mateIdx, state.controlledIdx);
            }
          }
        } else if (!teamHasBall && me.z === 0) {
          attemptSteal(state.controlledIdx);
        }
      } else if (action === 'switch') {
        if (!hasBall) {
          const mateIdx = state.controlledIdx === 0 ? 1 : 0;
          if (state.players[mateIdx] && !state.players[mateIdx].fouledOut) {
            state.controlledIdx = mateIdx;
            playTone(460, 'sine', 0.07, 0.09, 600);
          }
        }
      }
    }

    function launchAlleyOop(passerIdx, finisherIdx) {
      const passer = state.players[passerIdx];
      const finisher = state.players[finisherIdx];
      if (!passer || !finisher) return;

      const isPlayerSide = passer.side === 'player';
      const hoopX = isPlayerSide ? CONFIG.COURT.RIGHT_HOOP_X : CONFIG.COURT.LEFT_HOOP_X;
      const hoopY = CONFIG.COURT.HOOP_Y;
      const finisherDistToHoop = Math.hypot(hoopX - finisher.x, hoopY - finisher.y);

      const maxAlleyDist = CONFIG.DUNK_RANGE + 55;
      if (finisherDistToHoop > maxAlleyDist || finisher.dunk < CONFIG.DUNK_MIN_RATING) {
        launchPass(passerIdx, finisherIdx);
        if (finisherDistToHoop > maxAlleyDist) {
          triggerAnnouncer('TOO FAR FOR ALLEY-OOP!', 0.95);
        }
        return;
      }

      const catchX = hoopX + (isPlayerSide ? -68 : 68);
      const catchY = hoopY + (finisher.y >= hoopY ? 10 : -10);
      const lobDur = Math.max(0.42, Math.min(0.65, Math.hypot(catchX - passer.x, catchY - passer.y) / 920));

      finisher.facing = hoopX >= finisher.x ? 1 : -1;
      finisher.vz = CONFIG.DUNK_JUMP_VEL * 0.96;
      finisher.vx = (catchX - finisher.x) / lobDur;
      finisher.vy = (catchY - finisher.y) / lobDur;
      finisher.dunkType = finisher.dunkStyle || 'windmill';
      finisher.dunkTimer = lobDur + 0.45;

      state.ball.owner = -1;
      state.ball.inFlight = true;
      state.ball.flightType = 'alley_lob';
      state.ball.shooterIdx = finisherIdx;
      state.ball.startX = passer.x;
      state.ball.startY = passer.y;
      state.ball.startZ = passer.z + 52;
      state.ball.targetX = catchX;
      state.ball.targetY = catchY;
      state.ball.targetZ = 104;
      state.ball.flightTime = 0;
      state.ball.flightDuration = lobDur;

      if (passer.side === 'player') {
        state.controlledIdx = finisherIdx;
      }

      spawnParticles(passer.x, passer.y - 25, 10, '#38bdf8', 12, 4);
      triggerAnnouncer('ALLEY-OOP LOB!', 1.0);
      playTone(480, 'sine', 0.14, 0.12, 760);
    }

    function launchPass(fromIdx, toIdx) {
      const passer = state.players[fromIdx];
      const receiver = state.players[toIdx];
      if (!passer || !receiver) return;

      state.ball.owner = -1;
      state.ball.inFlight = true;
      state.ball.flightType = 'pass';
      state.ball.shooterIdx = toIdx;
      state.ball.startX = passer.x;
      state.ball.startY = passer.y;
      state.ball.startZ = passer.z + 45;
      state.ball.targetX = receiver.x;
      state.ball.targetY = receiver.y;
      state.ball.targetZ = 45;
      state.ball.flightTime = 0;
      state.ball.flightDuration = Math.max(0.18, Math.min(0.42, Math.hypot(receiver.x - passer.x, receiver.y - passer.y) / 1100));

      if (passer.side === 'player') {
        state.controlledIdx = toIdx;
      }
      playTone(390, 'sine', 0.08, 0.1, 520);
    }

    function attemptSteal(defenderIdx) {
      const defP = state.players[defenderIdx];
      if (!defP || defP.stunTimer > 0) return;

      defP.stunTimer = 0.32;
      defP.vx += defP.facing * 120;
      playTone(310, 'square', 0.07, 0.08, 220);

      if (state.ball.inFlight || state.ball.owner < 0 || state.graceTimer > 0) return;
      const handler = state.players[state.ball.owner];
      if (!handler || handler.side === defP.side) return;

      const dist = Math.hypot(handler.x - defP.x, handler.y - defP.y);
      if (dist < 64) {
        const stealChance = 0.34 + ((defP.def - 75) * 0.006);
        if (Math.random() < stealChance) {
          handler.knockdownTimer = 0.55;
          handler.knockdownDir = handler.x >= defP.x ? 1 : -1;
          handler.vx = handler.knockdownDir * 140;

          state.ball.owner = defenderIdx;
          if (defP.side === 'player') state.controlledIdx = defenderIdx;
          state.graceTimer = 0.75;
          spawnParticles(handler.x, handler.y, 14, '#facc15', 14, 4);
          triggerAnnouncer('PICKPOCKET! STOLEN!', 1.15);
          playTone(620, 'triangle', 0.14, 0.14, 880);
        }
      }
    }

    function triggerBlockJump(blockerIdx) {
      const blocker = state.players[blockerIdx];
      if (!blocker || blocker.z > 0 || blocker.stunTimer > 0) return;

      const defScale = 0.94 + ((blocker.def || 78) / 100) * 0.22;
      blocker.vz = CONFIG.SHOT_JUMP_VEL * defScale;
      blocker.blockingTimer = 0.68;
      blocker.swatSwing = 0;

      const targetX = state.ball.inFlight
        ? state.ball.x
        : (state.ball.owner >= 0 ? state.players[state.ball.owner].x : blocker.x);
      const targetY = state.ball.inFlight
        ? state.ball.y
        : (state.ball.owner >= 0 ? state.players[state.ball.owner].y : blocker.y);

      const dx = targetX - blocker.x;
      const dy = targetY - blocker.y;
      const dist = Math.hypot(dx, dy);
      if (dist > 4 && dist < 155) {
        blocker.vx += (dx / dist) * 155;
        blocker.vy += (dy / dist) * 155;
        blocker.facing = dx >= 0 ? 1 : -1;
      }
      playTone(320, 'triangle', 0.11, 0.11, 470);
    }

    function launchShot(shooterIdx) {
      const shooter = state.players[shooterIdx];
      if (!shooter) return;

      const isPlayerSide = shooter.side === 'player';
      const hoopX = isPlayerSide ? CONFIG.COURT.RIGHT_HOOP_X : CONFIG.COURT.LEFT_HOOP_X;
      const hoopY = CONFIG.COURT.HOOP_Y;
      const distToHoop = Math.hypot(hoopX - shooter.x, hoopY - shooter.y);
      const isThree = distToHoop > CONFIG.THREE_PT_RADIUS;
      const isDunk = !isThree && distToHoop <= CONFIG.DUNK_RANGE && shooter.dunk >= CONFIG.DUNK_MIN_RATING;

      let minDefDist = 999;
      let nearestDef = null;
      state.players.forEach(opp => {
        if (!opp.fouledOut && opp.side !== shooter.side && opp.knockdownTimer <= 0) {
          const d = Math.hypot(opp.x - shooter.x, opp.y - shooter.y);
          if (d < minDefDist) {
            minDefDist = d;
            nearestDef = opp;
          }
        }
      });

      const teamStreak = isPlayerSide ? state.playerStreak : state.cpuStreak;
      const isOnFire = teamStreak >= CONFIG.ON_FIRE_REQ;
      const isHeatingUp = teamStreak >= CONFIG.HEATING_UP_REQ;

      let makeProb = 0.45;
      if (isDunk) {
        makeProb = CONFIG.SHOT_TUNING.dunkBase + ((shooter.dunk - 75) * 0.005);
      } else if (distToHoop < 155) {
        makeProb = CONFIG.SHOT_TUNING.layupBase + ((shooter.dunk - 75) * 0.004);
      } else if (!isThree) {
        makeProb = CONFIG.SHOT_TUNING.midBase + ((shooter.three - 70) * 0.006);
      } else {
        makeProb = CONFIG.SHOT_TUNING.threeBase + ((shooter.three - 70) * 0.007);
      }

      if (isOnFire) makeProb += CONFIG.SHOT_TUNING.onFireBoost;
      else if (isHeatingUp) makeProb += 0.08;

      if (minDefDist < 95) {
        const contestFactor = (1 - minDefDist / 95) * CONFIG.SHOT_TUNING.contestPenaltyMax;
        makeProb -= contestFactor;
      }

      makeProb = Math.max(0.14, Math.min(0.95, makeProb));
      const willMake = Math.random() < makeProb;

      shooter.facing = hoopX >= shooter.x ? 1 : -1;

      if (isDunk) {
        shooter.vz = CONFIG.DUNK_JUMP_VEL;
        shooter.dunkType = shooter.dunkStyle || 'tomahawk';
        shooter.dunkTimer = 0.92;

        const flightDur = 0.86;
        const rimFrontX = hoopX + (isPlayerSide ? -18 : 18);
        shooter.vx = (rimFrontX - shooter.x) / flightDur;
        shooter.vy = (hoopY - shooter.y) / flightDur;

        state.ball.owner = -1;
        state.ball.inFlight = true;
        state.ball.flightType = 'dunk';
        state.ball.shooterIdx = shooterIdx;
        state.ball.points = 2;
        state.ball.willMake = willMake;
        state.ball.startX = shooter.x;
        state.ball.startY = shooter.y;
        state.ball.startZ = 54;
        state.ball.targetX = hoopX;
        state.ball.targetY = hoopY;
        state.ball.targetZ = 108;
        state.ball.flightTime = 0;
        state.ball.flightDuration = flightDur;

        const dunkCallouts = ['BOOMSHAKALAKA!', 'MONSTER JAM!', 'THROWN DOWN!', 'RIM ROCKER!'];
        triggerAnnouncer(dunkCallouts[Math.floor(Math.random() * dunkCallouts.length)], 1.25);
        playTone(220, 'sawtooth', 0.24, 0.14, 520);
      } else {
        shooter.vz = CONFIG.SHOT_JUMP_VEL;
        const flightDur = Math.max(0.55, Math.min(0.95, 0.5 + distToHoop / 950));

        state.ball.owner = -1;
        state.ball.inFlight = true;
        state.ball.flightType = 'shot';
        state.ball.shooterIdx = shooterIdx;
        state.ball.points = isThree ? 3 : 2;
        state.ball.willMake = willMake;
        state.ball.startX = shooter.x;
        state.ball.startY = shooter.y;
        state.ball.startZ = 62;
        state.ball.targetX = hoopX + (willMake ? 0 : (Math.random() < 0.5 ? -18 : 18));
        state.ball.targetY = hoopY + (willMake ? 0 : (Math.random() - 0.5) * 16);
        state.ball.targetZ = 108;
        state.ball.flightTime = 0;
        state.ball.flightDuration = flightDur;

        if (isThree) {
          triggerAnnouncer('FROM DOWNTOWN!', 1.05);
        }
        playTone(460, 'sine', 0.12, 0.11, 640);
      }

      if (nearestDef && nearestDef.side === 'cpu' && nearestDef.z === 0 && minDefDist < 130) {
        const jumpChance = state.cpuDiff === 'legend' ? 0.82 : (state.cpuDiff === 'allstar' ? 0.58 : 0.3);
        if (Math.random() < jumpChance) {
          triggerBlockJump(nearestDef.idx + 2);
        }
      }
    }

    function checkAerialBlockCollision() {
      const b = state.ball;
      if (!b.inFlight || (b.flightType !== 'shot' && b.flightType !== 'dunk')) return;
      const shooter = state.players[b.shooterIdx];
      if (!shooter) return;

      const t = b.flightDuration > 0 ? (b.flightTime / b.flightDuration) : 0;
      const isDescendingShot = b.flightType === 'shot' && t > 0.56 && t <= 0.96;

      if (!isDescendingShot && b.flightTime > b.flightDuration * 0.72) return;

      let goaltendTriggered = false;

      state.players.forEach(defP => {
        if (goaltendTriggered || defP.fouledOut || defP.side === shooter.side || defP.blockingTimer <= 0 || defP.z < 18) return;
        const distXY = Math.hypot(defP.x - b.x, defP.y - b.y);
        const distZ = Math.abs((defP.z + 55) - b.z);

        if (distXY < CONFIG.BLOCK_RANGE && distZ < CONFIG.BLOCK_Z_TOLERANCE) {
          // If the shot is on its downward trajectory near the cylinder, allow occasional goaltend grabs
          if (isDescendingShot) {
            const goaltendChance = 0.28 + ((defP.def - 75) * 0.004);
            if (Math.random() < goaltendChance) {
              goaltendTriggered = true;
              const defIdx = state.players.indexOf(defP);
              defP.blockingTimer = 0.15;
              defP.swatSwing = 1;

              b.inFlight = false;
              b.flightType = null;
              b.owner = defIdx;
              if (defP.side === 'player') {
                state.controlledIdx = defIdx;
              }
              state.graceTimer = 1.0;

              const isShooterPlayer = shooter.side === 'player';
              if (isShooterPlayer) {
                state.playerScore += b.points;
                state.playerStreak++;
              } else {
                state.cpuScore += b.points;
                state.cpuStreak++;
              }

              spawnParticles(b.x, b.y - b.z * 0.5, 16, '#facc15', 14, 4);
              triggerAnnouncer('GOALTENDING! COUNT IT!', 1.35);
              playTone(290, 'sawtooth', 0.22, 0.15, 440);
              updateHud();
            }
            return;
          }

          const posterChance = b.flightType === 'dunk'
            ? Math.max(0.18, Math.min(0.65, (shooter.dunk - defP.def + 20) * 0.015))
            : 0;

          if (b.flightType === 'dunk' && Math.random() < posterChance) {
            defP.blockingTimer = 0;
            defP.knockdownTimer = 0.75;
            defP.knockdownDir = defP.x >= shooter.x ? 1 : -1;
            defP.vx = defP.knockdownDir * 170;
            spawnParticles(defP.x, defP.y - 30, 18, '#fb923c', 18, 5);
            triggerAnnouncer('POSTERIZED! FACIAL SLAM!', 1.35);
            playTone(180, 'sawtooth', 0.22, 0.16, 95);
          } else {
            defP.blockingTimer = 0.2;
            defP.swatSwing = 1;
            if (defP.side === 'player') state.blocksCount++;

            b.flightType = 'rebound';
            b.shooterIdx = -1;
            b.startX = b.x;
            b.startY = b.y;
            b.startZ = b.z;
            b.targetX = b.x + defP.facing * (140 + Math.random() * 90);
            b.targetY = Math.max(CONFIG.COURT.TOP_Y + 20, Math.min(CONFIG.COURT.BOTTOM_Y - 20, b.y + (Math.random() - 0.5) * 140));
            b.targetZ = 18;
            b.flightTime = 0;
            b.flightDuration = 0.55;

            state.shakeTimer = 0.18;
            spawnParticles(b.x, b.y - b.z * 0.5, 20, '#38bdf8', 18, 5);
            triggerAnnouncer('GET THAT OUTTA HERE!', 1.3);
            playTone(150, 'square', 0.2, 0.16, 75);
          }
        }
      });
    }

    function resolveBasket() {
      const b = state.ball;
      const shooter = state.players[b.shooterIdx];
      const isPlayerSide = shooter ? shooter.side === 'player' : true;
      const wasDunk = b.flightType === 'dunk';

      if (b.willMake) {
        if (isPlayerSide) {
          state.playerScore += b.points;
          if (wasDunk) state.dunksCount++;
          state.playerStreak++;

          if (state.playerStreak === CONFIG.ON_FIRE_REQ) {
            triggerAnnouncer('YOUR SQUAD IS ON FIRE!', 1.5);
          } else if (state.playerStreak === CONFIG.HEATING_UP_REQ) {
            triggerAnnouncer('HEATING UP!', 1.25);
          }
        } else {
          state.cpuScore += b.points;
          state.cpuStreak++;

          if (state.cpuStreak === CONFIG.ON_FIRE_REQ) {
            triggerAnnouncer('CPU IS ON FIRE!', 1.5);
          } else if (state.cpuStreak === CONFIG.HEATING_UP_REQ) {
            triggerAnnouncer('CPU IS HEATING UP!', 1.25);
          }
        }

        if (wasDunk) {
          state.shakeTimer = 0.26;
          if (isPlayerSide) state.rimFlexRight = 1.0;
          else state.rimFlexLeft = 1.0;
          spawnParticles(b.targetX, CONFIG.COURT.HOOP_Y - 50, 28, '#facc15', 24, 5);
          playTone(130, 'sawtooth', 0.28, 0.18, 65);
        } else {
          spawnParticles(b.targetX, CONFIG.COURT.HOOP_Y - 50, 16, '#fb923c', 16, 4);
          playTone(660, 'sine', 0.14, 0.14, 880);
        }

        // Baseline inbound pass after every made basket
        let inbounderIdx = isPlayerSide ? 3 : 1;
        let receiverIdx = isPlayerSide ? 2 : 0;

        if (!isPlayerSide) {
          const activeUserIndices = [0, 1].filter(i => state.players[i] && !state.players[i].fouledOut);
          if (activeUserIndices.length === 1) {
            // Shorthanded 1-on-2: lone remaining player inbounds to themselves off the baseline
            const loneIdx = activeUserIndices[0];
            const lonePlayer = state.players[loneIdx];
            lonePlayer.x = CONFIG.COURT.LEFT_X + 36;
            lonePlayer.y = CONFIG.COURT.HOOP_Y;
            lonePlayer.z = 0;
            lonePlayer.vx = 0;
            lonePlayer.vy = 0;
            lonePlayer.vz = 0;
            lonePlayer.facing = 1;

            b.owner = loneIdx;
            b.inFlight = false;
            b.flightType = null;
            b.x = lonePlayer.x + 16;
            b.y = lonePlayer.y + 4;
            b.z = 36;

            state.controlledIdx = loneIdx;
            state.graceTimer = 1.25;
            updateHud();
            return;
          }
        }

        const inbounder = state.players[inbounderIdx];
        const receiver = state.players[receiverIdx];
        const baselineX = isPlayerSide ? CONFIG.COURT.RIGHT_X - 8 : CONFIG.COURT.LEFT_X + 8;
        const inboundCourtX = isPlayerSide ? CONFIG.COURT.RIGHT_X - 115 : CONFIG.COURT.LEFT_X + 115;

        inbounder.x = baselineX;
        inbounder.y = CONFIG.COURT.HOOP_Y + 42;
        inbounder.z = 0;
        inbounder.vx = 0;
        inbounder.vy = 0;
        inbounder.vz = 0;
        inbounder.facing = isPlayerSide ? -1 : 1;

        receiver.x = inboundCourtX;
        receiver.y = CONFIG.COURT.HOOP_Y - 28;
        receiver.z = 0;
        receiver.vx = 0;
        receiver.vy = 0;
        receiver.vz = 0;
        receiver.facing = isPlayerSide ? -1 : 1;

        b.owner = -1;
        b.inFlight = true;
        b.flightType = 'pass';
        b.shooterIdx = receiverIdx;
        b.startX = inbounder.x;
        b.startY = inbounder.y;
        b.startZ = 48;
        b.targetX = receiver.x;
        b.targetY = receiver.y;
        b.targetZ = 45;
        b.flightTime = 0;
        b.flightDuration = 0.36;

        if (!isPlayerSide) {
          state.controlledIdx = receiverIdx;
        }
        state.graceTimer = 1.25;
      } else {
        if (isPlayerSide) {
          if (state.playerStreak >= CONFIG.HEATING_UP_REQ) triggerAnnouncer('COOLED DOWN!', 1.1);
          state.playerStreak = 0;
        } else {
          if (state.cpuStreak >= CONFIG.HEATING_UP_REQ) triggerAnnouncer('CPU COOLED DOWN!', 1.1);
          state.cpuStreak = 0;
        }

        playTone(210, 'square', 0.14, 0.12, 140);
        b.flightType = 'rebound';
        b.shooterIdx = -1;
        b.startX = b.targetX;
        b.startY = b.targetY;
        b.startZ = 105;
        b.targetX = b.targetX + (isPlayerSide ? -110 - Math.random() * 90 : 110 + Math.random() * 90);
        b.targetY = Math.max(CONFIG.COURT.TOP_Y + 25, Math.min(CONFIG.COURT.BOTTOM_Y - 25, CONFIG.COURT.HOOP_Y + (Math.random() - 0.5) * 170));
        b.targetZ = 16;
        b.flightTime = 0;
        b.flightDuration = 0.62;
      }

      updateHud();
    }

    // 8. MOVEMENT, CPU AI & FRAME PHYSICS UPDATE
    // =========================================================================
    function handleHumanMovement(dt) {
      const me = state.players[state.controlledIdx];
      if (!me || me.knockdownTimer > 0 || me.z > 0) return;

      let ix = 0, iy = 0;
      if (keys['KeyW'] || keys['ArrowUp']) iy -= 1;
      if (keys['KeyS'] || keys['ArrowDown']) iy += 1;
      if (keys['KeyA'] || keys['ArrowLeft']) ix -= 1;
      if (keys['KeyD'] || keys['ArrowRight']) ix += 1;

      if (ix !== 0 || iy !== 0) {
        const len = Math.hypot(ix, iy);
        ix /= len;
        iy /= len;

        const wantTurbo = (keys['ShiftLeft'] || keys['ShiftRight']) && state.playerTurbo > 1;
        const speedStatMult = 0.86 + (me.speed / 100) * 0.26;
        const mult = wantTurbo ? CONFIG.TURBO_MULT : 1.0;
        const targetVx = ix * CONFIG.BASE_SPEED * speedStatMult * mult;
        const targetVy = iy * CONFIG.BASE_SPEED * speedStatMult * mult;

        const accel = wantTurbo ? 24 : 20;
        me.vx += (targetVx - me.vx) * Math.min(1, accel * dt);
        me.vy += (targetVy - me.vy) * Math.min(1, accel * dt);
        if (Math.abs(ix) > 0.1) me.facing = ix > 0 ? 1 : -1;

        if (wantTurbo) {
          state.playerTurbo = Math.max(0, state.playerTurbo - CONFIG.TURBO_DRAIN_RATE * dt);
        } else {
          state.playerTurbo = Math.min(100, state.playerTurbo + CONFIG.TURBO_REGEN_RATE * dt);
        }
      } else {
        me.vx *= Math.pow(0.0008, dt);
        me.vy *= Math.pow(0.0008, dt);
        state.playerTurbo = Math.min(100, state.playerTurbo + CONFIG.TURBO_REGEN_RATE * dt);
      }
    }

    function updateAiPlayers(dt) {
      const diffMult = state.cpuDiff === 'legend' ? 1.14 : (state.cpuDiff === 'allstar' ? 0.99 : 0.82);

      state.players.forEach((p, idx) => {
        if (p.fouledOut) return;
        if (p.side === 'player' && idx === state.controlledIdx) return;
        if (p.knockdownTimer > 0 || p.z > 0) return;

        let tx = p.x, ty = p.y;
        const isPlayerSide = p.side === 'player';
        const attackHoopX = isPlayerSide ? CONFIG.COURT.RIGHT_HOOP_X : CONFIG.COURT.LEFT_HOOP_X;
        const defendHoopX = isPlayerSide ? CONFIG.COURT.LEFT_HOOP_X : CONFIG.COURT.RIGHT_HOOP_X;
        const hoopY = CONFIG.COURT.HOOP_Y;

        if (state.ball.inFlight && state.ball.flightType === 'rebound') {
          tx = state.ball.targetX;
          ty = state.ball.targetY;
        } else if (!state.ball.inFlight && state.ball.owner === idx) {
          const driveTargetX = attackHoopX + (isPlayerSide ? -155 : 155);
          const driveTargetY = hoopY + (idx % 2 === 0 ? -65 : 65);
          tx = driveTargetX;
          ty = driveTargetY;

          const distToHoop = Math.hypot(attackHoopX - p.x, hoopY - p.y);
          const inRange = distToHoop < (p.three >= 87 ? 355 : 220);
          if (inRange && state.graceTimer <= 0 && Math.random() < 0.032 * diffMult) {
            launchShot(idx);
            return;
          }
          const mateIdx = idx === 2 ? 3 : 2;
          const mate = state.players[mateIdx];
          if (mate && Math.hypot(attackHoopX - mate.x, hoopY - mate.y) < distToHoop - 90 && Math.random() < 0.025) {
            launchPass(idx, mateIdx);
            return;
          }
        } else if (!state.ball.inFlight && state.ball.owner >= 0) {
          const holder = state.players[state.ball.owner];
          if (holder.side === p.side) {
            const spotX = attackHoopX + (isPlayerSide ? -240 : 240);
            const spotY = idx % 2 === 0 ? 205 : 525;
            tx = spotX;
            ty = spotY;
          } else {
            const oppPlayers = state.players.filter(o => o.side !== p.side && !o.fouledOut);
            const mark = oppPlayers[idx % oppPlayers.length] || holder;
            tx = mark.x * 0.72 + defendHoopX * 0.28;
            ty = mark.y * 0.78 + hoopY * 0.22;

            if (p.side === 'cpu' && mark === holder && Math.hypot(holder.x - p.x, holder.y - p.y) < 52 && Math.random() < 0.018 * diffMult) {
              attemptSteal(idx);
            }
          }
        }

        const dx = tx - p.x;
        const dy = ty - p.y;
        const dist = Math.hypot(dx, dy);
        if (dist > 8) {
          const spd = CONFIG.BASE_SPEED * (p.side === 'cpu' ? 0.88 * diffMult : 0.92);
          p.vx += ((dx / dist) * spd - p.vx) * Math.min(1, 10 * dt);
          p.vy += ((dy / dist) * spd - p.vy) * Math.min(1, 10 * dt);
          if (Math.abs(dx) > 4) p.facing = dx > 0 ? 1 : -1;
        } else {
          p.vx *= 0.85;
          p.vy *= 0.85;
        }
      });
    }

    function updatePhysics(dt) {
      if (state.graceTimer > 0) state.graceTimer = Math.max(0, state.graceTimer - dt);
      if (state.rimFlexLeft > 0) state.rimFlexLeft = Math.max(0, state.rimFlexLeft - dt * 2.4);
      if (state.rimFlexRight > 0) state.rimFlexRight = Math.max(0, state.rimFlexRight - dt * 2.4);
      if (state.shakeTimer > 0) state.shakeTimer = Math.max(0, state.shakeTimer - dt);

      if (state.announcerTimer > 0) {
        state.announcerTimer -= dt;
        if (state.announcerTimer <= 0) {
          document.getElementById('announcerBanner').classList.add('hidden');
        }
      }

      state.clock -= dt;
      state.mathTimer -= dt;

      if (state.clock <= 0) {
        state.clock = 0;
        state.mathTimer = 0;
        updateHud();
        state.pendingQuarterEnd = true;
        triggerMathTimeout();
        return;
      }

      if (state.mathTimer <= 0) {
        triggerMathTimeout();
        return;
      }

      handleHumanMovement(dt);
      updateAiPlayers(dt);

      state.players.forEach(p => {
        if (p.fouledOut) return;
        if (p.stunTimer > 0) p.stunTimer = Math.max(0, p.stunTimer - dt);
        if (p.knockdownTimer > 0) p.knockdownTimer = Math.max(0, p.knockdownTimer - dt);
        if (p.blockingTimer > 0) {
          p.blockingTimer = Math.max(0, p.blockingTimer - dt);
          p.swatSwing = Math.min(1, p.swatSwing + dt * 4.5);
        }
        if (p.dunkTimer > 0) p.dunkTimer = Math.max(0, p.dunkTimer - dt);

        const timeScale = (p.dunkTimer > 0 && p.z > 25) ? 0.62 : 1.0;

        p.x += p.vx * dt * timeScale;
        p.y += p.vy * dt * timeScale;

        if (p.z > 0 || p.vz > 0) {
          p.z += p.vz * dt * timeScale;
          p.vz -= CONFIG.GRAVITY * dt * timeScale;
          if (p.z <= 0) {
            p.z = 0;
            p.vz = 0;
            p.dunkType = null;
            p.dunkTimer = 0;
          }
        }

        p.x = Math.max(CONFIG.COURT.LEFT_X + 22, Math.min(CONFIG.COURT.RIGHT_X - 22, p.x));
        p.y = Math.max(CONFIG.COURT.TOP_Y + 18, Math.min(CONFIG.COURT.BOTTOM_Y - 12, p.y));

        const moveSpd = Math.hypot(p.vx, p.vy);
        if (p.z === 0 && moveSpd > 15) {
          p.runPhase += moveSpd * dt * 0.055;
        }
      });

      const b = state.ball;
      if (b.inFlight) {
        b.flightTime += dt;
        const t = Math.min(1, b.flightTime / b.flightDuration);

        if (b.flightType === 'dunk') {
          const dunker = state.players[b.shooterIdx];
          if (dunker) {
            b.x = dunker.x + dunker.facing * 18;
            b.y = dunker.y;
            b.z = dunker.z + (t < 0.55 ? 74 : 46);
          }
        } else {
          b.x = b.startX + (b.targetX - b.startX) * t;
          b.y = b.startY + (b.targetY - b.startY) * t;
          const arcPeak = b.flightType === 'shot' ? 145 : (b.flightType === 'alley_lob' ? 115 : (b.flightType === 'rebound' ? 75 : 35));
          b.z = b.startZ + (b.targetZ - b.startZ) * t + Math.sin(t * Math.PI) * arcPeak;
        }

        checkAerialBlockCollision();

        if (t >= 1 && b.inFlight) {
          if (b.flightType === 'pass') {
            b.inFlight = false;
            b.flightType = null;
            b.owner = b.shooterIdx;
          } else if (b.flightType === 'alley_lob') {
            const finisher = state.players[b.shooterIdx];
            const isPlayerSide = finisher ? finisher.side === 'player' : true;
            const hoopX = isPlayerSide ? CONFIG.COURT.RIGHT_HOOP_X : CONFIG.COURT.LEFT_HOOP_X;
            const hoopY = CONFIG.COURT.HOOP_Y;
            const distFromCatch = finisher ? Math.hypot(finisher.x - b.targetX, finisher.y - b.targetY) : 999;

            if (!finisher || finisher.knockdownTimer > 0 || distFromCatch > 75) {
              b.flightType = 'rebound';
              b.shooterIdx = -1;
              b.startX = b.x;
              b.startY = b.y;
              b.startZ = b.z;
              b.targetX = hoopX + (isPlayerSide ? -60 : 60);
              b.targetY = hoopY + (Math.random() - 0.5) * 90;
              b.targetZ = 16;
              b.flightTime = 0;
              b.flightDuration = 0.5;
              triggerAnnouncer('ERRANT LOB!', 1.0);
              playTone(210, 'square', 0.14, 0.11, 140);
            } else {
              const teamStreak = isPlayerSide ? state.playerStreak : state.cpuStreak;
              let makeProb = CONFIG.SHOT_TUNING.dunkBase + 0.08 + ((finisher.dunk - 75) * 0.004);
              if (teamStreak >= CONFIG.ON_FIRE_REQ) makeProb += CONFIG.SHOT_TUNING.onFireBoost;
              else if (teamStreak >= CONFIG.HEATING_UP_REQ) makeProb += 0.08;
              makeProb = Math.max(0.35, Math.min(0.96, makeProb));

              const slamDur = 0.28;
              const rimFrontX = hoopX + (isPlayerSide ? -16 : 16);
              finisher.vx = (rimFrontX - finisher.x) / slamDur;
              finisher.vy = (hoopY - finisher.y) / slamDur;
              finisher.vz = Math.max(finisher.vz, 120);
              finisher.dunkTimer = slamDur + 0.2;

              b.flightType = 'dunk';
              b.points = 2;
              b.willMake = Math.random() < makeProb;
              b.startX = b.x;
              b.startY = b.y;
              b.startZ = b.z;
              b.targetX = hoopX;
              b.targetY = hoopY;
              b.targetZ = 108;
              b.flightTime = 0;
              b.flightDuration = slamDur;

              spawnParticles(b.x, b.y - b.z * 0.5, 16, '#facc15', 16, 5);
              triggerAnnouncer('ALLEY-OOP SLAM!', 1.35);
              playTone(260, 'sawtooth', 0.2, 0.15, 620);
            }
          } else if (b.flightType === 'shot' || b.flightType === 'dunk') {
            resolveBasket();
          } else if (b.flightType === 'rebound') {
            let bestIdx = 0;
            let bestDist = 9999;
            state.players.forEach((pl, i) => {
              if (!pl.fouledOut && pl.knockdownTimer <= 0) {
                const d = Math.hypot(pl.x - b.x, pl.y - b.y);
                if (d < bestDist) {
                  bestDist = d;
                  bestIdx = i;
                }
              }
            });
            b.inFlight = false;
            b.flightType = null;
            b.owner = bestIdx;
            if (state.players[bestIdx].side === 'player') {
              state.controlledIdx = bestIdx;
            }
            playTone(420, 'sine', 0.08, 0.09);
          }
        }
      } else if (b.owner >= 0 && state.players[b.owner] && !state.players[b.owner].fouledOut) {
        const holder = state.players[b.owner];
        const dribbleBounce = holder.z === 0 ? Math.abs(Math.sin(holder.runPhase * 1.4)) * 22 : 0;
        b.x = holder.x + holder.facing * 16;
        b.y = holder.y + 4;
        b.z = holder.z + 36 - dribbleBounce;
      }

      for (let i = state.particles.length - 1; i >= 0; i--) {
        const pt = state.particles[i];
        pt.x += pt.vx * dt;
        pt.y += pt.vy * dt;
        pt.vy += 260 * dt;
        pt.life -= dt;
        if (pt.life <= 0) state.particles.splice(i, 1);
      }

      updateHud();
    }

    function renderBall() {
      const b = state.ball;
      const bx = b.x;
      const by = b.y - b.z * 0.88;
      let ballOnFire = false;
      if (!b.inFlight && b.owner >= 0 && state.players[b.owner]) {
        const ownerSide = state.players[b.owner].side;
        ballOnFire = ownerSide === 'player'
          ? state.playerStreak >= CONFIG.ON_FIRE_REQ
          : state.cpuStreak >= CONFIG.ON_FIRE_REQ;
      } else if (b.inFlight && b.shooterIdx >= 0 && state.players[b.shooterIdx]) {
        const shooterSide = state.players[b.shooterIdx].side;
        ballOnFire = shooterSide === 'player'
          ? state.playerStreak >= CONFIG.ON_FIRE_REQ
          : state.cpuStreak >= CONFIG.ON_FIRE_REQ;
      }

      ctx.fillStyle = 'rgba(0, 0, 0, 0.3)';
      ctx.beginPath();
      ctx.ellipse(bx, b.y + 3, 9, 4.5, 0, 0, Math.PI * 2);
      ctx.fill();

      if (ballOnFire) {
        const flameGrad = ctx.createRadialGradient(bx, by, 4, bx, by, 22);
        flameGrad.addColorStop(0, 'rgba(254, 240, 138, 0.95)');
        flameGrad.addColorStop(0.5, 'rgba(249, 115, 22, 0.75)');
        flameGrad.addColorStop(1, 'rgba(239, 68, 68, 0)');
        ctx.fillStyle = flameGrad;
        ctx.beginPath();
        ctx.arc(bx, by, 22, 0, Math.PI * 2);
        ctx.fill();
      }

      ctx.fillStyle = ballOnFire ? '#fb923c' : '#ea580c';
      ctx.strokeStyle = '#111827';
      ctx.lineWidth = 1.6;
      ctx.beginPath();
      ctx.arc(bx, by, 8.5, 0, Math.PI * 2);
      ctx.fill();
      ctx.stroke();

      ctx.beginPath();
      ctx.moveTo(bx - 8, by);
      ctx.lineTo(bx + 8, by);
      ctx.moveTo(bx, by - 8);
      ctx.lineTo(bx, by + 8);
      ctx.stroke();
    }

    function renderParticles() {
      state.particles.forEach(pt => {
        ctx.globalAlpha = Math.max(0, pt.life / pt.maxLife);
        ctx.fillStyle = pt.color;
        ctx.beginPath();
        ctx.arc(pt.x, pt.y, pt.size, 0, Math.PI * 2);
        ctx.fill();
      });
      ctx.globalAlpha = 1;
    }

    function render() {
      const cw = canvas.width;
      const ch = canvas.height;
      ctx.clearRect(0, 0, cw, ch);

      const scale = Math.min(cw / VIRTUAL_W, ch / VIRTUAL_H);
      const offsetX = (cw - VIRTUAL_W * scale) * 0.5;
      const offsetY = (ch - VIRTUAL_H * scale) * 0.5;

      ctx.save();
      ctx.translate(offsetX, offsetY);
      ctx.scale(scale, scale);

      if (state.shakeTimer > 0) {
        const mag = state.shakeTimer * 26;
        ctx.translate((Math.random() - 0.5) * mag, (Math.random() - 0.5) * mag);
      }

      renderCourt();
      renderHoop(true);
      renderHoop(false);

      const sorted = state.players.map((p, idx) => ({ p, idx })).sort((a, b) => a.p.y - b.p.y);
      sorted.forEach(item => renderPlayer(item.p, item.idx));

      renderBall();
      renderParticles();

      ctx.restore();
    }

    // =========================================================================
    // 10. MAIN ANIMATION LOOP & EVENT LISTENERS
    // =========================================================================
    let lastTimestamp = 0;
    function gameLoop(timestamp) {
      if (!lastTimestamp) lastTimestamp = timestamp;
      const dt = Math.min(0.05, (timestamp - lastTimestamp) / 1000);
      lastTimestamp = timestamp;

      if (state.mode === 'PLAYING') {
        updatePhysics(dt);
      }
      render();
      requestAnimationFrame(gameLoop);
    }

    window.addEventListener('keydown', (e) => {
      if (document.activeElement && document.activeElement.id === 'teamSearchInput') {
        if (e.code === 'Escape') {
          document.getElementById('teamSelectModal').classList.add('hidden');
        }
        return;
      }

      if (['Space', 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', 'Tab'].includes(e.code)) {
        e.preventDefault();
      }
      keys[e.code] = true;

      if (state.mode === 'PLAYING' && !e.repeat) {
        if (e.code === 'Space') handlePlayerAction('shoot');
        else if (e.code === 'KeyE') handlePlayerAction('pass');
        else if (e.code === 'KeyF' || e.code === 'Tab') handlePlayerAction('switch');
        else if (e.code === 'KeyP') togglePause();
      } else if (state.mode === 'PAUSED' && !e.repeat && e.code === 'KeyP') {
        togglePause();
      } else if (state.mode === 'MATH_TIMEOUT' && !e.repeat && state.mathAnswered) {
        if (e.code === 'Enter' || e.code === 'Space') {
          resumeFromMathTimeout();
        }
      } else if (e.code === 'Escape' && !document.getElementById('teamSelectModal').classList.contains('hidden')) {
        document.getElementById('teamSelectModal').classList.add('hidden');
      }
    });

    window.addEventListener('keyup', (e) => {
      keys[e.code] = false;
    });

    window.addEventListener('blur', () => {
      Object.keys(keys).forEach(k => { keys[k] = false; });
    });

    }
